Copy this SQL → paste → Run All

Do not type it by hand. Click the Copy button, paste into the worksheet, then run.

SQL · Lab 00 bootstrap (safe to re-run) 
-- Lab 00: bootstrap playground (safe to re-run)
-- Run as ACCOUNTADMIN (or equivalent) first.

USE ROLE ACCOUNTADMIN;

CREATE WAREHOUSE IF NOT EXISTS WH_ARCH_XS
  WAREHOUSE_SIZE = 'XSMALL'
  AUTO_SUSPEND = 60
  AUTO_RESUME = TRUE
  INITIALLY_SUSPENDED = TRUE
  COMMENT = 'Interview prep lab warehouse';

CREATE ROLE IF NOT EXISTS ROLE_ARCH_LAB_ADMIN
  COMMENT = 'Owns interview prep lab objects';

GRANT USAGE ON WAREHOUSE WH_ARCH_XS TO ROLE ROLE_ARCH_LAB_ADMIN;
GRANT OPERATE ON WAREHOUSE WH_ARCH_XS TO ROLE ROLE_ARCH_LAB_ADMIN;

CREATE DATABASE IF NOT EXISTS ARCH_INTERVIEW
  COMMENT = 'Snowflake Architect interview prep';

GRANT OWNERSHIP ON DATABASE ARCH_INTERVIEW TO ROLE ROLE_ARCH_LAB_ADMIN COPY CURRENT GRANTS;

SET uname = (SELECT CURRENT_USER());
GRANT ROLE ROLE_ARCH_LAB_ADMIN TO USER IDENTIFIER($uname);

USE ROLE ROLE_ARCH_LAB_ADMIN;
USE WAREHOUSE WH_ARCH_XS;
USE DATABASE ARCH_INTERVIEW;

CREATE SCHEMA IF NOT EXISTS RAW;
CREATE SCHEMA IF NOT EXISTS CURATED;
CREATE SCHEMA IF NOT EXISTS SECURE;
CREATE SCHEMA IF NOT EXISTS UTILS;

SELECT CURRENT_ROLE() AS role, CURRENT_WAREHOUSE() AS wh, CURRENT_DATABASE() AS db;
SHOW SCHEMAS IN DATABASE ARCH_INTERVIEW;
How to run in Snowsight Click inside the worksheet → press Ctrl+Enter (Windows) or click the blue Run button. If there are many statements, use Run All / run the whole script.
If the GRANT ROLE line fails That just means Snowflake wants your username typed. Run this tiny query first:
SQL · find your username 
SELECT CURRENT_USER();

Then run (replace the name with yours):

SQL · grant lab role to you 
USE ROLE ACCOUNTADMIN;
GRANT ROLE ROLE_ARCH_LAB_ADMIN TO USER "PASTE_YOUR_USERNAME_HERE";

Load sample customers (needed for Lab 01)

Tiny fake table with emails and SSN last-4 so we can practice hiding PII next.

SQL · sample customers 
USE ROLE ROLE_ARCH_LAB_ADMIN;
USE WAREHOUSE WH_ARCH_XS;
USE DATABASE ARCH_INTERVIEW;
USE SCHEMA RAW;

CREATE OR REPLACE TABLE CUSTOMERS (
  CUSTOMER_ID   NUMBER,
  FULL_NAME     STRING,
  EMAIL         STRING,
  PHONE         STRING,
  COUNTRY       STRING,
  SSN_LAST4     STRING,
  CREATED_AT    TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP()
);

INSERT INTO CUSTOMERS (CUSTOMER_ID, FULL_NAME, EMAIL, PHONE, COUNTRY, SSN_LAST4) VALUES
  (1, 'Asha Patel',  'asha.patel@example.com',  '+1-555-0101', 'US', '1234'),
  (2, 'Ben Okonkwo', 'ben.okonkwo@example.com', '+1-555-0102', 'US', '5678'),
  (3, 'Chen Wei',    'chen.wei@example.com',    '+1-555-0103', 'CA', '9012'),
  (4, 'Diana Rossi', 'diana.rossi@example.com', '+1-555-0104', 'US', '3456'),
  (5, 'Evan Ng',     'evan.ng@example.com',     '+1-555-0105', 'UK', '7890');

SELECT * FROM CUSTOMERS ORDER BY CUSTOMER_ID;
